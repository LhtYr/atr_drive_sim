# ============================================================
# 涡轮标定参数文件（依据真实试验数据 参数.md，7 个工况点）
#
# 注意：Turbine.c 的编译期默认值已直接采用本文件的标定值，
# 默认运行（make run / build/char，不带 --config）即为标定涡轮。
# 本文件保留作为标定值的显式记录，用于：
#   ① 代码默认值日后变更时恢复标定基准；
#   ② 作为 --config 参数文件的格式示例。
#
# 用法（与主仿真 --config / 特性工具 -c 同格式）：
#   ./build/main --config turbine_params.txt
#   python tools/char.py turbine -c turbine_params.txt --plot
#   校验：make char && ./build/char turbine-check
#
# 标定依据与误差量级（详见 README 4.7 节）：
#   - gamma/gas_R 取 7 点算术均值附近的圆整值（逐点 1.175~1.183 / 374~385）；
#     壅塞流量对 gamma、R 仅弱敏感（~sqrt(gamma/R)），
#     固定 gamma=1.18、gas_R=380 相对逐点真实物性的流量误差 < 0.8%，
#     逐点物性下反算 Cd_nozzle ≈ 1.001，故取 1.0。
#   - A_nozzle 为试验给定喉部流通面积 0.00246 m^2。
#   - r_choke_nozzle=0.7：试验第 7 点（膨胀比 1.5，压比 0.667）流量仍与
#     壅塞公式吻合（收扩喷嘴特性），故将壅塞压比上限从经典临界压比
#     （约 0.57）上调至 0.7。
#   - U/C 效率多项式与中径、机械效率取自试验报告：
#     eta = -3.266·(u/c0)^2 + 2.469·(u/c0) + 0.051，中径 110 mm，eta_mech 0.98。
# ============================================================

turbine.gamma          = 1.18
turbine.gas_R          = 380.0
turbine.A_nozzle       = 0.00246
turbine.Cd_nozzle      = 1.0
turbine.r_choke_nozzle = 0.7

turbine.mean_radius    = 0.055
turbine.eta_mech       = 0.98
turbine.eta_uc_a       = -3.266
turbine.eta_uc_b       = 2.469
turbine.eta_uc_c       = 0.051
